# =============================================================================
# AHORA — consola SSH (Bitvise)
# Arregla el error #1044 de phpMyAdmin
# =============================================================================

sudo mysql

# Dentro de mysql pega esto y Enter:

CREATE DATABASE IF NOT EXISTS DecoCakeShop
  CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
GRANT ALL PRIVILEGES ON DecoCakeShop.* TO 'usercodex'@'localhost';
FLUSH PRIVILEGES;
SHOW DATABASES;
EXIT;

# Después recarga phpMyAdmin (F5). Debes ver DecoCakeShop.
# NO importes los .sql de SQL Server en phpMyAdmin.

# =============================================================================
# 1) DNS — Linode Cloud Manager (navegador)
# =============================================================================
# Domains → usercodex.com → Add an A record:
#   Hostname: decocakeshop
#   IP:       198.58.117.160
# Eso crea: https://decocakeshop.usercodex.com
# La etiqueta del Linode (decocakeshop) y el hostname DNS coinciden.

# =============================================================================
# 2) Código en el servidor — consola SSH
# =============================================================================
# Opción A: si el repo ya está en GitHub (cambia la URL):

cd ~
git clone https://github.com/TU_USUARIO/DecoCakeShop.git decocakeshop_src

# Opción B: si aún no hay GitHub, sube el proyecto por SFTP
#   Local:  D:\Startup\DecoCakeShop
#   Remoto: /home/usercodex/decocakeshop_src
#   (incluye Backend y Frontend; no subas Backend/.venv ni Frontend/node_modules)

# =============================================================================
# 3) .env de producción — consola SSH
# =============================================================================

grep DB_PASSWORD ~/academia_src/Backend/.env

cd ~/decocakeshop_src/Backend
cp deploy/env.production.example .env
nano .env

# En nano: pega esa contraseña en DB_PASSWORD
# SECRET_KEY: cambia el texto
# Ctrl+O Enter  Ctrl+X

# =============================================================================
# 4) Instalar backend + cargar MySQL — consola SSH
# =============================================================================
# Sube primero Backend/db_scripts_mysql/ (ya está en tu PC) junto con el resto
# del proyecto. No importes nada en phpMyAdmin.

cd ~/decocakeshop_src/Backend
chmod +x scripts/bootstrap_server.sh
./scripts/bootstrap_server.sh

# =============================================================================
# 5) Gunicorn — consola SSH
# =============================================================================

sudo cp ~/decocakeshop_src/Backend/deploy/gunicorn-decocakeshop.service /etc/systemd/system/
sudo systemctl daemon-reload
sudo systemctl enable gunicorn-decocakeshop
sudo systemctl start gunicorn-decocakeshop
sudo systemctl status gunicorn-decocakeshop --no-pager

# =============================================================================
# 6) Frontend — PowerShell EN TU PC (no en Linode)
# =============================================================================

cd D:\Startup\DecoCakeShop\Frontend
npm install
npm run build

# SFTP: sube Frontend/dist/*  →  /home/usercodex/decocakeshop_front/

# En SSH, si la carpeta no existe:
mkdir -p ~/decocakeshop_front

# =============================================================================
# 7) Apache — consola SSH
# =============================================================================

sudo cp ~/decocakeshop_src/Backend/deploy/decocakeshop.usercodex.com.conf /etc/apache2/sites-available/
sudo a2enmod proxy proxy_http rewrite
sudo a2ensite decocakeshop.usercodex.com.conf
sudo chown -R usercodex:www-data /home/usercodex/decocakeshop_front
sudo chown -R usercodex:www-data /home/usercodex/decocakeshop_src/Backend/staticfiles
sudo apache2ctl configtest && sudo systemctl reload apache2

# =============================================================================
# 8) HTTPS — consola SSH
# =============================================================================

sudo certbot --apache -d decocakeshop.usercodex.com

# Si HTTPS da 404 (Certbot a veces deja el :443 sin DocumentRoot):
chmod 755 /home/usercodex
sudo cp ~/decocakeshop_src/Backend/deploy/decocakeshop.usercodex.com-le-ssl.conf /etc/apache2/sites-available/
sudo apache2ctl configtest && sudo systemctl reload apache2

# =============================================================================
# Si algo falla
# =============================================================================
sudo journalctl -u gunicorn-decocakeshop -n 80 --no-pager
sudo tail -n 50 /var/log/apache2/decocakeshop_error.log
